Data Encryption
DynamoDB
fields @timestamp, @logStream, @log
| filter @message like /gpg.exe/
| parse @message 'CommandLine: *CurrentDirectory' as CommandLine
| parse @message 'CurrentDirectory: *User' as CurrentDirectory
| parse @message 'User: *LogonGuid' as User
| parse @message 'Hashes: *ParentProcessGuid' as Hashes
| parse @message 'ParentCommandLine: *ParentUser' as ParentCommandLine
| parse @message 'Host Application = *Engine Version' as HostApplication
| parse @message 'Command Name = *Command Type' as CommandName
| sort @timestamp desc
